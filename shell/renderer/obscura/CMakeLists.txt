cmake_minimum_required(VERSION 3.20)
project(weber_obscura_boundary LANGUAGES CXX)
# Independent renderer target; intentionally not linked to electron_lib/V8.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WEBER_ENGINE_LIBRARY "" CACHE FILEPATH "Built Obscura C ABI library")
if(NOT EXISTS "${WEBER_ENGINE_LIBRARY}")
  message(FATAL_ERROR "Set WEBER_ENGINE_LIBRARY to the built weber-engine static library")
endif()
add_library(weber_engine STATIC IMPORTED)
set_target_properties(weber_engine PROPERTIES IMPORTED_LOCATION "${WEBER_ENGINE_LIBRARY}")
find_package(Threads REQUIRED)
add_executable(weber-obscura-smoke obscura_engine.cc smoke.cc)
target_compile_options(weber-obscura-smoke PRIVATE -Wall -Wextra -Werror)
target_link_libraries(weber-obscura-smoke PRIVATE weber_engine Threads::Threads ${CMAKE_DL_LIBS} m rt util)
enable_testing()
add_test(NAME real-obscura-boundary COMMAND weber-obscura-smoke)
set_tests_properties(real-obscura-boundary PROPERTIES TIMEOUT 45)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(obscura_wire STATIC ../../common/obscura/wire.cc)
  add_library(obscura_browser_proxy STATIC ../../browser/obscura/renderer_process.cc)
  target_link_libraries(obscura_browser_proxy PUBLIC obscura_wire Threads::Threads)
  add_executable(weber-obscura-renderer obscura_engine.cc renderer_main.cc)
  target_link_libraries(weber-obscura-renderer PRIVATE weber_engine obscura_wire Threads::Threads ${CMAKE_DL_LIBS} m rt util)
  add_executable(weber-process-smoke ../../browser/obscura/process_smoke.cc)
  target_link_libraries(weber-process-smoke PRIVATE obscura_browser_proxy)
  add_executable(weber-renderer-fixture ../../browser/obscura/renderer_fixture.cc)
  target_link_libraries(weber-renderer-fixture PRIVATE obscura_wire)
  foreach(target obscura_wire obscura_browser_proxy weber-obscura-renderer weber-process-smoke weber-renderer-fixture)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endforeach()
  add_test(NAME renderer-transport-failures COMMAND weber-process-smoke --fake $<TARGET_FILE:weber-renderer-fixture>)
  add_test(NAME real-renderer-processes COMMAND weber-process-smoke --real $<TARGET_FILE:weber-obscura-renderer>)
  set_tests_properties(renderer-transport-failures PROPERTIES TIMEOUT 15)
  set_tests_properties(real-renderer-processes PROPERTIES TIMEOUT 90)
endif()
