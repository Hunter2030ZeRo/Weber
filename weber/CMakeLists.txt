cmake_minimum_required(VERSION 3.20)
project(weber_desktop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
# This runtime build has no Chromium checkout, GN, Content or Blink dependency.
add_subdirectory(../shell/renderer/obscura obscura)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
find_package(nlohmann_json 3.2 REQUIRED)
find_package(X11 REQUIRED)
find_package(Threads REQUIRED)
add_library(weber_desktop_output STATIC ../shell/browser/obscura/desktop/async_output.cc)
target_compile_options(weber_desktop_output PRIVATE -Wall -Wextra -Werror)
target_link_libraries(weber_desktop_output PUBLIC Threads::Threads)
add_executable(weber-desktop-host ../shell/browser/obscura/desktop/main.cc ../shell/browser/obscura/desktop/frame.cc ../shell/browser/obscura/desktop/platform_sync.cc
  ../shell/browser/obscura/desktop/display.cc ../shell/browser/obscura/desktop/notification.cc ../shell/browser/obscura/desktop/clipboard.cc ../shell/browser/obscura/desktop/resource_broker.cc)
target_compile_options(weber-desktop-host PRIVATE -Wall -Wextra -Werror)
target_link_libraries(weber-desktop-host PRIVATE obscura_browser_proxy weber_desktop_output PkgConfig::GTK nlohmann_json::nlohmann_json X11::X11)
add_executable(weber-async-output-test tests/async-output.cc)
target_compile_options(weber-async-output-test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(weber-async-output-test PRIVATE weber_desktop_output)
add_test(NAME desktop-output-backpressure COMMAND weber-async-output-test)
add_executable(weber-frame-test tests/frame.cc ../shell/browser/obscura/desktop/frame.cc)
target_compile_options(weber-frame-test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(weber-frame-test PRIVATE PkgConfig::GTK)
add_test(NAME desktop-frame-pixels COMMAND weber-frame-test)
set_tests_properties(desktop-output-backpressure PROPERTIES TIMEOUT 10)
add_executable(weber-shortcut-focus tests/shortcut-focus.cc)
target_compile_options(weber-shortcut-focus PRIVATE -Wall -Wextra -Werror)
target_link_libraries(weber-shortcut-focus PRIVATE X11::X11)
